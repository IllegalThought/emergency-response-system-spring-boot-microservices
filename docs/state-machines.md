# Campus Naksha State Machines


# 1. Issue State Machine

Initial state:

REPORTED


Allowed transitions:

REPORTED -> ACKNOWLEDGED

REPORTED -> REJECTED

ACKNOWLEDGED -> IN_PROGRESS

ACKNOWLEDGED -> REJECTED

IN_PROGRESS -> RESOLVED


Final states:

RESOLVED

REJECTED


Invalid examples:

RESOLVED -> IN_PROGRESS

REJECTED -> ACKNOWLEDGED

IN_PROGRESS -> REPORTED



# 2. Guard State Machine

Typical lifecycle:

OFFLINE
    ->
OFF_DUTY
    ->
AVAILABLE
    ->
RESERVED
    ->
BUSY
    ->
AVAILABLE


Allowed transitions:

OFFLINE -> OFF_DUTY

OFF_DUTY -> AVAILABLE

AVAILABLE -> RESERVED

RESERVED -> BUSY

RESERVED -> AVAILABLE

BUSY -> AVAILABLE

AVAILABLE -> OFF_DUTY

OFF_DUTY -> OFFLINE



# 3. SOS State Machine

Normal flow:

CREATED
    ->
SEARCHING_FOR_GUARD
    ->
GUARD_ASSIGNED
    ->
ACCEPTED
    ->
RESPONDING
    ->
RESOLVED


No guard flow:

CREATED
    ->
SEARCHING_FOR_GUARD
    ->
NO_GUARD_AVAILABLE


Allowed transitions:

CREATED -> SEARCHING_FOR_GUARD

CREATED -> CANCELLED

SEARCHING_FOR_GUARD -> GUARD_ASSIGNED

SEARCHING_FOR_GUARD -> NO_GUARD_AVAILABLE

SEARCHING_FOR_GUARD -> CANCELLED

GUARD_ASSIGNED -> ACCEPTED

GUARD_ASSIGNED -> SEARCHING_FOR_GUARD

GUARD_ASSIGNED -> CANCELLED

ACCEPTED -> RESPONDING

RESPONDING -> RESOLVED


Invalid examples:

RESOLVED -> RESPONDING

CANCELLED -> ACCEPTED

NO_GUARD_AVAILABLE -> RESOLVED
